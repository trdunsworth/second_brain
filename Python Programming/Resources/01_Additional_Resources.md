# Install essential packages
install.packages(c(
  "tidyverse", "tidymodels", "ggplot2",
  "data.table", "dtplyr", "arrow", "duckdb",
  "renv", "styler", "lintr", "testthat",
  "usethis", "devtools", "roxygen2", "pkgdown",
  "targets", "tarchetypes", "crew",
  "vetiver", "plumber", "pins",
  "shiny", "bslib", "thematic",
  "quarto", "rmarkdown", "bookdown",
  "leaflet", "mapview", "sf", "tmap",
  "plotly", "ggiraph", "echarts4r",
  "DT", "reactable", "gt", "flextable",
  "janitor", "skimr", "visdat", "naniar",
  "inspectdf", "DataExplorer", "DataExplorer"
))

# Quick setup
